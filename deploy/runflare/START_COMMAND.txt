# ============================================
# Runflare — دستورات درست (مهم)
# ============================================
#
# ۱) دستور اصلی / Start / Command  ← این را بگذار (ترجیحی):
bash /app/deploy/runflare/start.sh
#
# ۲) دستور پیش از شروع / Pre-start  ← اختیاری (اگر ۱ را گذاشتی):
bash /app/deploy/runflare/prestart.sh
#
# ۳) فیلد ماژول اپلیکیشن اگر خالی بود بگذار:
koohestan.asgi
#
# --------------------------------------------
# اگر فقط prestart داری و Start جدا نیست، همان prestart کافی است
# ولی فیلد ماژول را خالی نگذار — حتماً koohestan.asgi
#
# دستور gunicorn دستی (اگر لازم شد):
gunicorn koohestan.asgi:application -k uvicorn.workers.UvicornWorker --bind 0.0.0.0:8000 --workers 2 --timeout 120
